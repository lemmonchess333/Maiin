describe("zz run plans (scratch)", () => {
  it("writes the template catalogue", () => {
    writeFileSync(
      `${OUT}/_templates.json`,
      JSON.stringify(
        RUN_TEMPLATES.map((t) => ({
          id: t.id,
          name: t.name,
          type: t.type,
          minutes: t.estimatedDuration,
          config: t.config,
          description: t.description,
        })),
        null,
        1
      )
    );
    expect(true).toBe(true);
  });

  it.each(CASES.map((c) => [c.persona.name, c] as const))(
    "%s",
    (name, c) => {
      const season = simulateLiftSeason(c.persona, {
        weeks: c.weeks,
        seed: c.seed ?? 1,
        ...(c.perfect ? { runnerVariant: NO_INJURY } : {}),
      });
      const habits = c.persona.running!;
      const fitness = habits.benchmark
        ? {
            benchmark: habits.benchmark,
            vdot:
              Math.round(
                vdotFromRace(
                  habits.benchmark.distanceM,
                  habits.benchmark.timeS
                ) * 10
              ) / 10,
          }
        : null;
      const start = c.persona.start ?? DEFAULT_START;
      const out = {
        name,
        answers: c.persona.answers,
        runner: habits.runner,
        runShare: habits.runShare,
        benchmark: habits.benchmark ?? null,
        goalTimeS: habits.goalTimeS ?? null,
        paceTable: paceTableFromFitness(fitness),
        baseline: runnerStartingPoint(c.persona, start),
        breaks: c.persona.breaks ?? [],
        awayFromApp: c.persona.awayFromApp ?? false,
        runWeeks: season.runWeeks,
        runs: season.runs,
        events: season.events,
        failures: season.failures,
        coaching: runCoachingFindings(season, c.persona),
      };
      writeFileSync(`${OUT}/${name}.json`, JSON.stringify(out, null, 1));
      writeFileSync(`${OUT}/${name}.txt`, runTrace(season, c.persona));
      expect(season.runWeeks.length).toBeGreaterThan(0);
    },
    120_000
  );
});
