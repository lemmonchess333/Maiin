const DOW = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"];
const START = "2026-11-09"; // a Monday

function plusDays(date: string, n: number): string {
  return localDateString(addLocalDays(parseLocalDate(date), n));
}

function runInfo(rd: NonNullable<ProgramState["runDays"]>[number]) {
  const id = rd.userOverride || rd.templateId;
  const t = RUN_TEMPLATES.find((x) => x.id === id);
  return {
    id,
    type: rd.type,
    name: t?.name ?? id,
    minutes: t?.estimatedDuration ?? null,
    km: t?.config?.targetDistanceKm ?? null,
    clash: rd.clashesWithLift === true,
    status: rd.status,
    timeLimit: rd.timeLimit ?? null,
  };
}

function snapshot(
  label: string,
  state: ProgramState,
  weekSchedule: ScheduleDay[],
  weekStart: string,
  experience: string
) {
  const days = [];
  for (let i = 0; i < 7; i++) {
    const date = plusDays(weekStart, i);
    const dow = parseLocalDate(date).getDay();
    const li = liftIndexForDayOfWeek(weekSchedule, dow);
    const w = li >= 0 ? state.workouts[li] : undefined;
    const runs = (state.runDays ?? []).filter((rd) => rd.date === date);
    days.push({
      date,
      dow: DOW[dow],
      sched: weekSchedule[dow]?.type,
      lift: w
        ? {
            index: li,
            name: w.dayName,
            type: w.dayType,
            skipped: w.skipped === true,
            sets: w.exercises.reduce((t, e) => t + e.sets, 0),
            legSets: w.exercises
              .filter(loadsTheLegs)
              .reduce((t, e) => t + e.sets, 0),
            minutes: estimateSessionMinutes(w.exercises, {
              sessionMinutes: state.sessionMinutes,
            }),
            mains: w.exercises
              .filter((e) => e.isAccessory !== true)
              .map((e) => `${e.name} ${e.sets}x${e.reps}@${e.weight}`),
            legs: w.exercises
              .filter(loadsTheLegs)
              .map(
                (e) =>
                  `${e.name} ${e.sets}x${e.reps}@${e.weight}${e.isAccessory ? "" : "*"}`
              ),
            all: w.exercises.map(
              (e) =>
                `${e.name} ${e.sets}x${e.reps}@${e.weight}${e.isAccessory ? "" : "*"}`
            ),
            purpose: liftSessionPurpose(
              state,
              w,
              date,
              toExperience(experience)
            ),
            raceRestSkip: raceRestSkips(state, date).includes(li),
          }
        : null,
      runs: runs.map(runInfo),
    });
  }
  return {
    label,
    weekStart,
    phase: state.currentPhase,
    weekNumber: state.weekNumber,
    raceWeek: state.raceWeek ?? null,
    liftWeekKey: state.liftWeekKey,
    runPlan: state.runPlan
      ? {
          mode: state.runPlan.mode,
          currentWeek: state.runPlan.currentWeek,
          totalWeeks: state.runPlan.totalWeeks,
          phase: state.runPlan.phase,
          compressed: state.runPlan.compressed,
          belowFloor: state.runPlan.belowFloor,
          leadInWeek: state.runPlan.leadInWeek,
          stage: state.runPlan.stage,
        }
      : null,
    splitType: state.splitType,
    orphanRuns: (state.runDays ?? [])
      .filter((rd) => !rd.date || rd.date < weekStart || rd.date > plusDays(weekStart, 6))
      .map((rd) => ({ date: rd.date, ...runInfo(rd) })),
    days,
  };
}

it("prints the hybrid weeks", () => {
  mkdirSync("/tmp/work/train", { recursive: true });
  vi.useFakeTimers({ toFake: ["Date"] });
  const out: unknown[] = [];
  for (const p of PERSONAS) {
    vi.setSystemTime(new Date(`${START}T12:00:00`));
    const input = { ...base, ...p.input } as PlanBuilderInput;
    let plan;
    try {
      plan = buildPlan(input);
    } catch (err) {
      out.push({ persona: p.name, error: String(err) });
      continue;
    }
    let state = plan.programState;
    const createdAtMs = Date.now();
    const stamp = { toMillis: () => createdAtMs };
    const profile = {
      uid: "scratch",
      onboardingComplete: true,
      createdAt: stamp,
      onboardingCompletedAt: stamp,
      ...plan.profileUpdates,
      experience: input.experience,
      weightKg: input.bodyweightKg,
      gender: "male",
      runFrequency: p.runFrequency ?? "regular",
      athleteType: "Hybrid",
      ...(input.runMode === "starter"
        ? { starterGoal: { startedOn: START } }
        : {}),
    } as unknown as UserProfile;
    const weeks: unknown[] = [];
    let weekStart = localWeekKey(parseLocalDate(START));
    weeks.push(
      snapshot("week 1", state, plan.weekSchedule, weekStart, input.experience)
    );
    for (let w = 2; w <= p.weeks; w++) {
      // The person trains: every lift not skipped is done.
      state = {
        ...state,
        workouts: state.workouts.map((d) =>
          d.skipped ? d : { ...d, completed: true }
        ),
      };
      const today = plusDays(weekStart, 7);
      vi.setSystemTime(new Date(`${today}T12:00:00`));
      for (let pass = 0; pass < 20; pass++) {
        const run = rollRunWeeks(state, profile, today, "none", null);
        if (run.weeks > 0) {
          state = run.state;
          continue;
        }
        const lift = rollLiftWeeks(
          state,
          profile,
          localWeekKey(parseLocalDate(today))
        );
        if (lift.weeks > 0) {
          state = lift.state;
          continue;
        }
        break;
      }
      weekStart = today;
      weeks.push(
        snapshot(
          `week ${String(w)}`,
          state,
          plan.weekSchedule,
          weekStart,
          input.experience
        )
      );
    }
    out.push({
      persona: p.name,
      input: {
        liftDays: input.liftDays,
        weeklyRunDays: input.weeklyRunDays,
        runMode: input.runMode,
        primaryGoal: input.primaryGoal,
        raceGoal: input.raceGoal ?? null,
        raceLegTrim: input.raceLegTrim ?? null,
      },
      weekSchedule: plan.weekSchedule.map((d) => `${DOW[d.day]}:${d.type}`),
      profile: {
        runMode: plan.profileUpdates.runMode,
        weeklyRunDaysTarget: plan.profileUpdates.weeklyRunDaysTarget,
        weeklyWorkoutsTarget: plan.profileUpdates.weeklyWorkoutsTarget,
      },
      weeks,
    });
  }
  vi.useRealTimers();
  writeFileSync("/tmp/work/train/hybrid-weeks.json", JSON.stringify(out, null, 1));
});
