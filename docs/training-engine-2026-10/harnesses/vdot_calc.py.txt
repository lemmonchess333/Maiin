import math, sys

def vo2_cost(v):  # v in m/min -> mL/kg/min (Daniels & Gilbert 1979)
    return -4.60 + 0.182258*v + 0.000104*v*v

def frac_max(t):  # t in minutes -> fraction of VO2max sustainable
    return 0.8 + 0.1894393*math.exp(-0.012778*t) + 0.2989558*math.exp(-0.1932605*t)

def vdot(dist_m, t_min):
    v = dist_m / t_min
    return vo2_cost(v) / frac_max(t_min)

def race_time(vd, dist_m):
    lo, hi = 1.0, 2000.0
    for _ in range(200):
        mid = (lo+hi)/2
        if vdot(dist_m, mid) > vd: lo = mid  # too fast -> needs more time
        else: hi = mid
    return (lo+hi)/2

def v_at_vo2(target):
    # solve vo2_cost(v)=target
    a, b, c = 0.000104, 0.182258, -4.60 - target
    return (-b + math.sqrt(b*b - 4*a*c)) / (2*a)

def fmt(t_min):
    s = int(round(t_min*60)); h, r = divmod(s, 3600); m, s = divmod(r, 60)
    return f"{h}:{m:02d}:{s:02d}" if h else f"{m}:{s:02d}"

def pace(v):  # m/min -> min:sec per km
    p = 1000.0/v; m = int(p); s = int(round((p-m)*60))
    if s == 60: m, s = m+1, 0
    return f"{m}:{s:02d}/km"

D = {"5K":5000, "10K":10000, "HM":21097.5, "M":42195}

def table(vd):
    out = {k: race_time(vd, d) for k, d in D.items()}
    return out

    for k, d in D.items():

    for vd in [30, 40, 50]:
        mt = race_time(vd, 42195)
        row = []
