import fs from 'node:fs';import path from 'node:path';import {createHash}from'node:crypto';
import{cutoutWebp}from'../Maiin-burpees/scripts/art/keying';import{FORM_ARTWORK}from'../Maiin-burpees/src/lib/formArtwork';import{cardFrameOf}from'../Maiin-burpees/src/lib/formArtCardFrames';import{FORM_ART_CUTOUTS}from'../Maiin-burpees/src/lib/formArtCutouts.data';
const H=import.meta.dirname,R=path.resolve(H,'../Maiin-burpees'),D=path.join(R,'docs/exercise-art/reviews/2026-10-20-strict-review'),ids=JSON.parse(fs.readFileSync(path.join(D,'library-review.json'),'utf8')).correctedSets;
const sha=(b:Buffer)=>createHash('sha256').update(b).digest('hex');const rows=[];
for(const id of ids){const source=cardFrameOf(id,FORM_ARTWORK[id]),c=FORM_ART_CUTOUTS[id],bytes=fs.readFileSync(path.join(R,'public',source));const rebuilt=await cutoutWebp(path.join(R,'public',source));const delivered=fs.readFileSync(path.join(R,'public',c.src));if(source!==c.source||sha(bytes)!==c.sourceSha256||sha(delivered)!==c.sha256||!rebuilt.webp.equals(delivered))throw Error('Card mismatch '+id);rows.push({id,source,sourceSha256:sha(bytes),card:c.src,cardSha256:sha(delivered),normalCutoutReproductionByteExact:true,width:c.width,height:c.height});}
fs.writeFileSync(path.join(D,'final-card-binding.json'),JSON.stringify({scope:'Normal production cutoutWebp recipe, current cardFrameOf and public source/card bytes; no prepared override for these eight sets.',sets:rows},null,2)+'\n');console.log({cards:rows.length,allExact:true});
