def sum_window(a,k):
 p=np.pad(a,((1,0),(1,0))).cumsum(0).cumsum(1)
 return p[k:,k:]-p[:-k,k:]-p[k:,:-k]+p[:-k,:-k]
def snap(image,point,radius=18):
 x,y=map(round,point);x0=max(5,x-radius);y0=max(5,y-radius);x1=min(image.shape[1]-5,x+radius+1);y1=min(image.shape[0]-5,y+radius+1)
 a=image[y0-4:y1+4,x0-4:x1+4].mean(2);gy,gx=np.gradient(a);A=sum_window(gx*gx,9);B=sum_window(gy*gy,9);C=sum_window(gx*gy,9)
 score=(A+B-np.sqrt((A-B)**2+4*C*C))/2
 yy,xx=np.mgrid[:score.shape[0],:score.shape[1]];score=score/(1+((xx+x0-x)**2+(yy+y0-y)**2)/250)
 py,px=np.unravel_index(np.argmax(score),score.shape)
 return (int(px+x0),int(py+y0),float(score[py,px]))
def match(image,reference,point,radius=15,half=12):
 x,y=point[:2];target=reference[y-half:y+half+1,x-half:x+half+1].astype('int16')
 x0=max(half,x-radius);x1=min(image.shape[1]-half-1,x+radius);y0=max(half,y-radius);y1=min(image.shape[0]-half-1,y+radius)
 scores=[]
 for dy in range(y0,y1+1):
  strip=image[dy-half:dy+half+1,x0-half:x1+half+1].astype('int16')
  windows=np.lib.stride_tricks.sliding_window_view(strip,(2*half+1,2*half+1),axis=(0,1))[0].transpose(0,2,3,1)
  values=np.abs(windows-target).mean(axis=(1,2,3))
  scores.extend((float(value),xx,dy)for xx,value in zip(range(x0,x1+1),values))
 scores.sort();best=scores[0]
 distinct=next((v for v in scores if abs(v[1]-best[1])+abs(v[2]-best[2])>2),None)
 return {'x':best[1],'y':best[2],'meanAbsoluteRGBError':best[0],'distinctMatchGap':distinct[0]-best[0]if distinct else None}
