def palette(source,guard,pose,recipe):
    a=np.asarray(source);rgb=a[:,:,:3].astype(float);r,g,b=rgb[:,:,0],rgb[:,:,1],rgb[:,:,2];rules=recipe['sourceSupportRules']
    support=(b-r>=rules['bMinusRMin'])&(b-g>=rules['bMinusGMin'])&(r-g>=rules['rMinusGMin'])&(np.asarray(guard)>0)
    yy=np.indices(support.shape)[0];upper=support&(yy<=recipe['upperFieldSecondaryMaxY'])
    trace=Image.new('L',(SIZE[0]*4,SIZE[1]*4));d=ImageDraw.Draw(trace)
    for poly in recipe['posteriorThighSecondaryTraces'][str(pose)]:d.polygon([(x*4,y*4) for x,y in poly],fill=255)
    trace=trace.resize(SIZE,Image.Resampling.LANCZOS);ham=np.asarray(trace,dtype=float)/255;ham[~support]=0;ham[upper]=0
    u=upper.astype(float);primary=np.where(support,1.-u-ham,0.)
    maximum=np.max(rgb,axis=2);chroma=(maximum-g)/np.maximum(maximum,1);neutral=maximum[:,:,None]
    den=recipe['sourceTintStrengthDenominators']
    def mapped(target,key):
        k=np.minimum(chroma/den[key],1.)
        return neutral+(np.asarray(target)*maximum[:,:,None]/255-neutral)*k[:,:,None]
    out=mapped(recipe['primary'],'primary')*primary[:,:,None]+mapped(recipe['secondary'],'upperSecondary')*u[:,:,None]+mapped(recipe['secondary'],'hamstringSecondary')*ham[:,:,None]
    colour=np.zeros_like(a);colour[support,:3]=np.rint(np.clip(out[support],0,255)).astype(np.uint8);colour[support,3]=255
    proof={'existingSourceChromaPixels':int(support.sum()),'primaryPixelsAboveHalf':int(np.count_nonzero(primary>.5)),'upperSecondaryPixels':int(upper.sum()),'hamstringSecondaryPixelsAboveHalf':int(np.count_nonzero(ham>.5)),'neutralPixelsNotSelected':True,'noNewFieldOrBodyGradient':True,'sourceTonalCalibration':den,'roleSelection':'Existing source chroma only. PrimaryQuads+Glutes; secondaryFrontDelts+UpperBack, plus source4Hamstrings selected along the visible white fascia separation.','exactUncolouredSourceRecovered':False}
    masks={'source-colour-support':Image.fromarray((support*255).astype(np.uint8)),'primary-role':Image.fromarray(np.rint(primary*255).astype(np.uint8)),'secondary-role':Image.fromarray(np.rint((u+ham)*255).astype(np.uint8)),'hamstring-secondary-trace':trace}
    return Image.fromarray(colour),support,masks,proof

def ora(path,layers,expected):
    doc=ET.Element('image',w=str(expected.width),h=str(expected.height),name='Elliptical intact source + fixed equipment + source-tonal palette REVIEW')
    stack=ET.SubElement(doc,'stack');members={'mimetype':b'image/openraster','mergedimage.png':png(expected)}
    for i,(name,im) in reversed(list(enumerate(layers))):
        key=f'data/layer{i}.png';members[key]=png(im);ET.SubElement(stack,'layer',name=name,src=key,x='0',y='0',opacity='1.0',visibility='visible',**{'composite-op':'svg:src-over'})
    members['stack.xml']=ET.tostring(doc,encoding='utf-8')
    with zipfile.ZipFile(path,'x') as z:
        for name,data in members.items():
            entry=zipfile.ZipInfo(name,(2026,10,15,0,0,0));entry.compress_type=zipfile.ZIP_STORED if name=='mimetype' else zipfile.ZIP_DEFLATED;z.writestr(entry,data)
    return verify_ora(path,expected)

def verify_ora(path,expected):
    with zipfile.ZipFile(path) as z:
        assert z.testzip() is None and len(z.namelist())==len(set(z.namelist()))==8
        doc=ET.fromstring(z.read('stack.xml'));stack=doc.find('stack');out=Image.new('RGBA',(int(doc.attrib['w']),int(doc.attrib['h'])))
        for el in reversed(stack.findall('layer')):out=Image.alpha_composite(out,Image.open(BytesIO(z.read(el.attrib['src']))).convert('RGBA'))
        merged=Image.open(BytesIO(z.read('mergedimage.png'))).convert('RGBA')
        assert pixel(out)==pixel(merged)==pixel(expected)
    return {'path':path.name,'sha256':sha(path),'layers':5,'members':8,'stackAndMergedRgbaExact':True,'pixelSha256':pixel(expected)}
