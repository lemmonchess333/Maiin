def h(path):return sha256(path.read_bytes()).hexdigest()
def read(path):return json.loads(path.read_bytes())
def normal_format(data,path):
    package=read(REPO/'node_modules/prettier/package.json')
    cli=REPO/'node_modules/prettier'/(package['bin']['prettier'] if isinstance(package['bin'],dict) else package['bin'])
    return subprocess.check_output([str(NODE),str(cli),'--stdin-filepath',str(path)],input=data,cwd=REPO)
def dump(value):return (json.dumps(value,indent=2,ensure_ascii=False)+'\n').encode('utf-8')
def verify_run(folder):
    start=read(folder/'start.json');result=read(folder/'result.json')
    before=read(folder/'inputs-before.json');after=read(folder/'inputs-after.json')
    assert result['headBefore']==result['headAfter']==start['head']==BASE
    assert result['headUnchanged'] and result['statusRestored'] and result['allInputsUnchanged']
    assert result['statusBefore']==result['statusAfter']==start['status']
    assert before==after and len(before)==result['inputCount']==2656 and result['changedInputsAfterRestoration']==[]
    assert start['startedAt']==result['startedAt'] and result['finishedAt']>=result['startedAt']
    log=folder/result.get('log','test.log')
    if not log.is_file():log=folder/('verify.log' if result.get('normalCommand')=='npm run verify' else 'test.log')
    assert h(log)==result['logSha256']
    for i,name in enumerate(start['snapshotPaths']):
        capture=folder/f'snapshot-before-{i}.snap';assert capture.is_file()
        if name in before:assert h(capture)==before[name]
    for row in result['snapshotRestores']:
        i=start['snapshotPaths'].index(row['path'])
        assert row['restoredExact'] and h(folder/f'snapshot-before-{i}.snap')==row['originalSha256']==before[row['path']]
        assert h(folder/f'snapshot-test-output-{i}.snap')==row['testOutputSha256']
    return start,result,before,log
